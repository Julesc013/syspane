cmake_minimum_required(VERSION 3.28)
project(SysPane VERSION 0.0.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options(-Wall -Wextra -Wpedantic -Werror -march=x86-64 -mtune=generic)
if(WIN32)
    add_compile_definitions(WINVER=0x0A00 _WIN32_WINNT=0x0A00)
endif()
set(SYSPANE_PROFILE "" CACHE STRING "Pinned development profile")
if(NOT SYSPANE_PROFILE MATCHES "^(windows-x64-gcc15|linux-x64-gcc13)$")
    message(FATAL_ERROR "Select a checked-in development preset; no implicit supported target")
endif()
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/build-support/check_profile.py"
    "${SYSPANE_PROFILE}" "${CMAKE_CXX_COMPILER}" "${CMAKE_BINARY_DIR}" RESULT_VARIABLE profile_result)
if(NOT profile_result EQUAL 0)
    message(FATAL_ERROR "Development profile does not match this toolchain")
endif()

add_library(syspane_model STATIC source/model/state.cpp)
target_include_directories(syspane_model PUBLIC "${CMAKE_SOURCE_DIR}/source/model")
target_compile_options(syspane_model PRIVATE -Wall -Wextra -Wpedantic -Werror -march=x86-64 -mtune=generic)

add_executable(syspane_model_smoke source/application/model_smoke.cpp)
target_link_libraries(syspane_model_smoke PRIVATE syspane_model)
set_target_properties(syspane_model_smoke PROPERTIES OUTPUT_NAME "SysPane.ModelSmoke")

include(CTest)
if(BUILD_TESTING)
    add_executable(syspane_model_tests tests/model/model_tests.cpp)
    target_link_libraries(syspane_model_tests PRIVATE syspane_model)
    target_compile_options(syspane_model_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -march=x86-64 -mtune=generic)
    set(model_cases STATE-01 STATE-02 STATE-03 STATE-04 STATE-05 VALIDITY-01 VALIDITY-02
        CLOCK-01 CLOCK-02 BOUNDS-01 BOUNDS-02 BOUNDS-03 OBSERVATION-01 EPOCH-01 FRESHNESS-01)
    foreach(case IN LISTS model_cases)
        add_test(NAME "model.${case}" COMMAND syspane_model_tests "${case}")
    endforeach()
    add_test(NAME model.smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/model/check_smoke.py" "$<TARGET_FILE:syspane_model_smoke>")
    add_test(NAME composition.graph COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/build-support/check_components.py" "${CMAKE_BINARY_DIR}/component-graph.txt")
    add_test(NAME composition.reject_forbidden COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/build-support/check_components.py" "${CMAKE_BINARY_DIR}/component-graph.txt" --negative-test)
endif()

set(component_targets syspane_model syspane_model_smoke)
if(BUILD_TESTING)
    list(APPEND component_targets syspane_model_tests)
endif()
set(component_graph "")
foreach(component IN LISTS component_targets)
    string(APPEND component_graph "${component}|$<TARGET_PROPERTY:${component},TYPE>|$<TARGET_PROPERTY:${component},LINK_LIBRARIES>|$<TARGET_PROPERTY:${component},SOURCES>\n")
    if(WIN32 AND NOT component STREQUAL "syspane_model")
        target_link_options(${component} PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
endforeach()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/component-graph.txt" CONTENT "${component_graph}")
install(TARGETS syspane_model_smoke RUNTIME DESTINATION bin COMPONENT DevelopmentSmoke)
