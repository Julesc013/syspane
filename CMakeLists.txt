cmake_minimum_required(VERSION 3.28)
project(SysPane VERSION 0.0.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options(-Wall -Wextra -Wpedantic -Werror -march=x86-64 -mtune=generic)
if(WIN32)
    add_compile_definitions(WINVER=0x0A00 _WIN32_WINNT=0x0A00)
endif()
set(SYSPANE_PROFILE "" CACHE STRING "Pinned development profile")
if(NOT SYSPANE_PROFILE MATCHES "^(windows-x64-gcc15|linux-x64-gcc13)$")
    message(FATAL_ERROR "Select a checked-in development preset; no implicit supported target")
endif()
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/build-support/check_profile.py"
    "${SYSPANE_PROFILE}" "${CMAKE_CXX_COMPILER}" "${CMAKE_BINARY_DIR}" RESULT_VARIABLE profile_result)
if(NOT profile_result EQUAL 0)
    message(FATAL_ERROR "Development profile does not match this toolchain")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/build-support/check_dependencies.py"
    RESULT_VARIABLE dependency_result)
if(NOT dependency_result EQUAL 0)
    message(FATAL_ERROR "Vendored dependency verification failed")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_SOURCE_DIR}/spec/experience/settings-registry.json"
    "${CMAKE_SOURCE_DIR}/build-support/generate_settings.py"
    "${CMAKE_SOURCE_DIR}/build-support/dependencies.json")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/build-support/generate_settings.py"
    "${CMAKE_BINARY_DIR}/generated/settings_descriptors.hpp" RESULT_VARIABLE settings_result)
if(NOT settings_result EQUAL 0)
    message(FATAL_ERROR "Native settings descriptor generation failed")
endif()

add_library(syspane_json INTERFACE source/third_party/nlohmann/json.hpp)
target_include_directories(syspane_json SYSTEM INTERFACE "${CMAKE_SOURCE_DIR}/source/third_party")
target_compile_definitions(syspane_json INTERFACE JSON_DIAGNOSTIC_POSITIONS=1 JSON_DIAGNOSTICS=0)
add_library(syspane_protocol STATIC source/protocol/wire.cpp source/protocol/ledger.cpp source/protocol/outbox.cpp)
target_include_directories(syspane_protocol PUBLIC "${CMAKE_SOURCE_DIR}/source/protocol")
target_link_libraries(syspane_protocol PUBLIC syspane_json)
add_library(syspane_configuration STATIC source/configuration/policy.cpp source/configuration/session.cpp)
target_include_directories(syspane_configuration PUBLIC "${CMAKE_SOURCE_DIR}/source/configuration"
    PRIVATE "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(syspane_configuration PUBLIC syspane_protocol)

add_library(syspane_model STATIC source/model/state.cpp)
add_library(syspane_local_ipc STATIC source/platform/local_ipc_linux.cpp source/platform/local_ipc_windows.cpp)
target_include_directories(syspane_local_ipc PUBLIC "${CMAKE_SOURCE_DIR}/source/platform")
if(WIN32)
    target_link_libraries(syspane_local_ipc PRIVATE advapi32)
endif()
target_include_directories(syspane_model PUBLIC "${CMAKE_SOURCE_DIR}/source/model")
target_compile_options(syspane_model PRIVATE -Wall -Wextra -Wpedantic -Werror -march=x86-64 -mtune=generic)

add_executable(syspane_model_smoke source/application/model_smoke.cpp)
target_link_libraries(syspane_model_smoke PRIVATE syspane_model)
set_target_properties(syspane_model_smoke PROPERTIES OUTPUT_NAME "SysPane.ModelSmoke")
add_executable(syspane_ipc_probe source/application/ipc_probe.cpp)
target_link_libraries(syspane_ipc_probe PRIVATE syspane_local_ipc syspane_configuration syspane_protocol)
set_target_properties(syspane_ipc_probe PROPERTIES OUTPUT_NAME "SysPane.IpcProbe")

include(CTest)
if(BUILD_TESTING)
    add_executable(syspane_model_tests tests/model/model_tests.cpp)
    target_link_libraries(syspane_model_tests PRIVATE syspane_model)
    target_compile_options(syspane_model_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -march=x86-64 -mtune=generic)
    set(model_cases STATE-01 STATE-02 STATE-03 STATE-04 STATE-05 VALIDITY-01 VALIDITY-02
        CLOCK-01 CLOCK-02 BOUNDS-01 BOUNDS-02 BOUNDS-03 OBSERVATION-01 EPOCH-01 FRESHNESS-01)
    foreach(case IN LISTS model_cases)
        add_test(NAME "model.${case}" COMMAND syspane_model_tests "${case}")
    endforeach()
    add_test(NAME model.smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/model/check_smoke.py" "$<TARGET_FILE:syspane_model_smoke>")
    add_test(NAME composition.graph COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/build-support/check_components.py" "${CMAKE_BINARY_DIR}/component-graph.txt")
    add_test(NAME composition.reject_forbidden COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/build-support/check_components.py" "${CMAKE_BINARY_DIR}/component-graph.txt" --negative-test)
    add_executable(syspane_protocol_tests tests/protocol/protocol_tests.cpp)
    target_link_libraries(syspane_protocol_tests PRIVATE syspane_protocol syspane_configuration)
    set(protocol_cases FRAME-01 FRAME-02 FRAME-03 JSON-01 WIRE-01 NEGOTIATE-01
        IPC-BUDGET-01 IPC-BUDGET-02 IPC-BUDGET-03 IPC-BUDGET-04 LEDGER-01 POLICY-01 POLICY-02 DISCLOSURE-01 QUEUE-01 SESSION-01)
    foreach(case IN LISTS protocol_cases)
        add_test(NAME "protocol.${case}" COMMAND syspane_protocol_tests "${case}")
    endforeach()
    add_test(NAME protocol.dependencies COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/build-support/check_dependencies.py")
    foreach(family NATIVE-01 NATIVE-02)
        add_test(NAME "native.${family}" COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/protocol/native_ipc.py"
            "$<TARGET_FILE:syspane_ipc_probe>" "${family}" "${CMAKE_BINARY_DIR}/native-evidence")
        set_tests_properties("native.${family}" PROPERTIES TIMEOUT 90)
    endforeach()
endif()

set(component_targets syspane_model syspane_model_smoke syspane_json syspane_protocol syspane_configuration syspane_local_ipc syspane_ipc_probe)
if(BUILD_TESTING)
    list(APPEND component_targets syspane_model_tests syspane_protocol_tests)
endif()
set(component_graph "")
foreach(component IN LISTS component_targets)
    string(APPEND component_graph "${component}|$<TARGET_PROPERTY:${component},TYPE>|$<TARGET_PROPERTY:${component},LINK_LIBRARIES>|$<TARGET_PROPERTY:${component},SOURCES>\n")
    get_target_property(component_type ${component} TYPE)
    if(WIN32 AND component_type STREQUAL "EXECUTABLE")
        target_link_options(${component} PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
endforeach()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/component-graph.txt" CONTENT "${component_graph}")
install(TARGETS syspane_model_smoke RUNTIME DESTINATION bin COMPONENT DevelopmentSmoke)
